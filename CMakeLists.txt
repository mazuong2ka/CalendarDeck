cmake_minimum_required(VERSION 3.24)
project(CalendarDeck VERSION 0.2.4 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Calendar Deck requires Windows 10 1903 or later")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(deck_core src/calendar.cpp src/platform.cpp)
target_include_directories(deck_core PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/src>")
target_link_libraries(deck_core PUBLIC winhttp crypt32 shell32 ole32 advapi32)
add_executable(CalendarDeck WIN32 src/app.cpp resources/app.rc)
target_include_directories(CalendarDeck PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/resources>")
set_source_files_properties(resources/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/calendar-deck.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
configure_file(resources/app.manifest app.manifest COPYONLY)
configure_file(resources/calendar-deck.ico calendar-deck.ico COPYONLY)
configure_file(resources/resource.h resource.h COPYONLY)
target_link_libraries(CalendarDeck PRIVATE deck_core d2d1 dwrite comctl32 comdlg32)
if(MSVC)
  target_compile_options(deck_core PRIVATE /W4 /utf-8)
  target_compile_options(CalendarDeck PRIVATE /W4 /utf-8)
  set_property(TARGET deck_core CalendarDeck PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(deck_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(CalendarDeck PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(CalendarDeck PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(deck_tests tests/core_tests.cpp)
  target_link_libraries(deck_tests PRIVATE deck_core)
  if(NOT MSVC)
    target_link_options(deck_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME calendar_core COMMAND deck_tests)
endif()
